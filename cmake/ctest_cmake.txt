# Условная регистрация тестовых таргетов.
# Включить через: include(cmake/ctest_cmake.txt)
# Активировать: cmake ... -DPUF_BUILD_TESTS=ON
#
# Использование:
#   puf_add_test(
#       TARGET      sram_puf_test
#       SRCS        test/sram_puf_test.cpp
#       INCLUDE_DIRS test
#       LINK_LIBS   sram_puf
#   )

option(PUF_BUILD_TESTS "Build unit tests (requires Conan host build)" OFF)

function(puf_add_test)
    cmake_parse_arguments(ARG "" "TARGET" "SRCS;INCLUDE_DIRS;LINK_LIBS" ${ARGN})

    if(NOT PUF_BUILD_TESTS)
        return()
    endif()

    find_package(GTest REQUIRED)
    add_executable(${ARG_TARGET} ${ARG_SRCS})

    if(ARG_INCLUDE_DIRS)
        target_include_directories(${ARG_TARGET} PRIVATE ${ARG_INCLUDE_DIRS})
    endif()

    target_link_libraries(${ARG_TARGET} PRIVATE ${ARG_LINK_LIBS} GTest::gtest_main)
    add_test(NAME ${ARG_TARGET} COMMAND ${ARG_TARGET})
endfunction()
